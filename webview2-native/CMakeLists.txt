cmake_minimum_required(VERSION 3.24)
project(compose_webview2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only Windows x64 is supported")
endif()
find_package(JNI REQUIRED)
include(cmake/WebView2Sdk.cmake)
add_library(compose_webview2 SHARED src/jni_bridge.cpp src/webview_host.cpp)
target_compile_features(compose_webview2 PRIVATE cxx_std_20)
target_include_directories(compose_webview2 PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_definitions(compose_webview2 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(compose_webview2 PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(compose_webview2 PRIVATE JNI::AWT d3d11 windowsapp webview2_loader ole32 oleaut32 shlwapi version)
set_property(TARGET compose_webview2 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
install(TARGETS compose_webview2 RUNTIME DESTINATION native/windows-x64)
